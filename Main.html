timelist <- c("timetoreport", "timetoppsreportdate", "timetoppsmdrodate")

datetable <- mutate_at(screendata2, vars(contains("date")), mdy)

timedata <- datetable %>% 
  mutate(timetoreport = as.numeric(abs(difftime(datetable$report_date, datetable$mdro_date, units = "days"))),
         timetoppsreportdate =  as.numeric(abs(difftime(datetable$pps_date, datetable$report_date, units = "days"))),
         timetoppsmdrodate =  as.numeric(abs(difftime(datetable$pps_date, datetable$mdro_date, units = "days")))) %>% 
  drop_na(timelist)

allscreenAC <- timedata %>% 
  filter(setting == "Acute care hospital") %>% 
  mutate(setting_acuity = case_match(as.integer(setting_acuity),
                                   1 ~ "High acuity Unit",
                                   2 ~ "Not High acuity Unit",
                                   9 ~ "unknown")) %>% 
  group_by(setting, setting_acuity) %>% 
  summarise(n = n(),
            mediantests = median(as.integer(na.omit(pps_denom_p1))),
            IQRtests = IQR(as.integer(na.omit(pps_denom_p1))),
            mediantimetopps = median(as.integer(na.omit(timetoppsreportdate))),
            IQRtimetopps = IQR(as.integer(na.omit(timetoppsreportdate))))
`summarise()` has grouped output by 'setting'. You can override using the
`.groups` argument.
allscreenAC$setting_acuity <- str_c(allscreenAC$setting_acuity, allscreenAC$setting, sep = " ")

allscreenother <- timedata %>% 
  filter(!setting == "Acute care hospital") %>% 
  mutate(setting_acuity = case_when(setting_acuity >= 1 ~ setting,
                                    .default = setting)) %>% 
  group_by(setting_acuity) %>% 
  summarise(n = n(),
            mediantests = median(as.integer(na.omit(pps_denom_p1))),
            IQRtests = IQR(as.integer(na.omit(pps_denom_p1))),
            mediantimetopps = median(as.integer(na.omit(timetoppsreportdate))),
            IQRtimetopps = IQR(as.integer(na.omit(timetoppsreportdate))))

testsperfacilitytype <- rbind(allscreenAC, allscreenother) 

testsperfacilitytype[,-1] %>% arrange(desc(n))
